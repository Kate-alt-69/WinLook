cmake_minimum_required(VERSION 3.16)
project(wl_Lcpuv VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
add_library(wl_Lcpuv STATIC src/wl_Lcpuv.c)
add_library(WinLook::Lcpuv ALIAS wl_Lcpuv)
target_include_directories(wl_Lcpuv PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(wl_Lcpuv PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE YES)
install(TARGETS wl_Lcpuv ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/wl_Lcpuv.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
include(CTest)
if(BUILD_TESTING)
 add_executable(test_wl_Lcpuv tests/test_wl_Lcpuv.c)
 target_link_libraries(test_wl_Lcpuv PRIVATE wl_Lcpuv)
 add_test(NAME wl_Lcpuv.core COMMAND test_wl_Lcpuv)
endif()
