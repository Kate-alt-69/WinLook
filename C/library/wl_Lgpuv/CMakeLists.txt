cmake_minimum_required(VERSION 3.16)
project(wl_Lgpuv VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
add_library(wl_Lgpuv STATIC src/wl_Lgpuv.c)
add_library(WinLook::Lgpuv ALIAS wl_Lgpuv)
target_include_directories(wl_Lgpuv PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(wl_Lgpuv PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE YES)
install(TARGETS wl_Lgpuv ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/wl_Lgpuv.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
include(CTest)
if(BUILD_TESTING)
 add_executable(test_wl_Lgpuv tests/test_wl_Lgpuv.c)
 target_link_libraries(test_wl_Lgpuv PRIVATE wl_Lgpuv)
 add_test(NAME wl_Lgpuv.core COMMAND test_wl_Lgpuv)
endif()
